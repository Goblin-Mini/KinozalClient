cmake_minimum_required(VERSION 3.22.1)
project(ktvguard LANGUAGES CXX)
if(NOT DEFINED KTV_OFFICIAL_CERT_SHA256 OR KTV_OFFICIAL_CERT_SHA256 STREQUAL "")
    message(FATAL_ERROR "KTV_OFFICIAL_CERT_SHA256 must be supplied by Gradle")
endif()
add_library(ktvguard SHARED native_guard.cpp)
target_compile_features(ktvguard PRIVATE cxx_std_17)
target_compile_definitions(ktvguard PRIVATE KTV_OFFICIAL_CERT_SHA256="\${KTV_OFFICIAL_CERT_SHA256}")
target_compile_options(ktvguard PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -fvisibility-inlines-hidden -fstack-protector-strong -D_FORTIFY_SOURCE=2)
target_link_options(ktvguard PRIVATE -Wl,--gc-sections -Wl,-z,relro -Wl,-z,now -Wl,--exclude-libs,ALL)
